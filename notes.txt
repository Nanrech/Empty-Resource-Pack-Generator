--- CLIENT JAR ---
Example:
https://piston-data.mojang.com/v1/objects/e877b6a07acd633fb3bb475002175cec036e7b87/client.jar

Contains:
// A jar file called `client.jar`
// Note: This file can be renamed to client.zip and decompressed like a normal zip folder
/client
  /assets
    /minecraft
      /lang
        deprecated.json      // JSON object with 2 objects, "removed": [str] and "renamed": { str: str }
        en_us.json           // What we want
    .mcassetsroot            // Empty file
  /com
  /data
    /minecraft               // Apparently contains all the datapacks that make the game work
  /META-INF
  /net                       // All the code (a ton of .class files)
  flightrecorder-config.jfc
  pack.png
  version.json               // (str: int) "pack_version": {"resource_major", "resource_minor", "data_major", "data_minor"}

--- SERVER JAR ---
Example:
https://piston-data.mojang.com/v1/objects/33680f5f2ac32864d6d7cf5e56a705fdb3e05f4c/server.jar

Contains:
// A jar file called `server.jar`
// Note: This file can be renamed to server.jar and decompressed like a normal zip folder
/server
  /META-INF
    /libraries
    /versions
      /26.3
        server-26.3.jar                // Another jar!
        /decompressed                  // This folder has to be generated manually, same as the other jar files
          /assets
            /minecraft
              /lang
                deprecated.json        // Contains the exact same stuff as client/assets/minecraft/lang/deprecated.json
                en_us.json             // Contains the exact same stuff as client/assets/minecraft/lang/en_us.json
            .mcassetsroot
          /com
          /data
            /minecraft                 // Contains the exact same stuff as client/data/minecraft
          /META-INF
          /net
          flightrecorder-config.jfc
          version.json                 // Contains the exact same stuff as client/version.json
    classpath-joined
    libraries.list
    main-class
    MANIFEST.MF
    versions.list                      // tab separated file, contains "JAR_HASH	JAR_VERSION	JAR_PATH" (path to server-26.3.jar from /versions)
  /net
  version.json
